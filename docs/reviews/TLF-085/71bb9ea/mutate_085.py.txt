"""Opus TLF-085 review: my own mutations in the scratch copy (not Flash's in-suite ones). Each
mutation edits one file, runs the named test files (one Godot call each, hard 120 s cap) and
restores the file byte for byte. Usage: python mutate_085.py <copy dir>"""
import os, re, subprocess, sys, pathlib

COPY = pathlib.Path(sys.argv[1])
GODOT = r"D:\Godot\Godot_v4.7.2-stable_win64_console.exe"
SCENE = "scenes/chapters/ch01/rooms/maintenance_bay.tscn"
ROUTER = "scripts/rooms/room_router.gd"
BAY = "res://tests/chapters/ch01/test_ch01_maintenance_bay_v2.gd"
NEG = "res://tests/chapters/ch01/test_ch01_step1_negatives.gd"
PZ = "res://tests/chapters/ch01/test_ch01_pz002_negative.gd"


def disable(node_header):
    return (node_header, node_header + "\ndisabled = true")


MUTATIONS = [
    ("MV1 bulkhead north segment disabled", SCENE, *disable('[node name="CollisionShape2D_1" type="CollisionShape2D" parent="Bulkhead"]'), [BAY, NEG]),
    ("MV2 bulkhead south segment disabled", SCENE, *disable('[node name="CollisionShape2D_2" type="CollisionShape2D" parent="Bulkhead"]'), [BAY, NEG]),
    ("MV3 shutter divider north disabled", SCENE, *disable('[node name="DividerNorth" type="CollisionShape2D" parent="ShutterDividers"]'), [BAY, PZ]),
    ("MV4 shutter divider south disabled", SCENE, *disable('[node name="DividerSouth" type="CollisionShape2D" parent="ShutterDividers"]'), [BAY, PZ]),
    ("MV5 inner divider north disabled", SCENE, *disable('[node name="CollisionShape2D_1" type="CollisionShape2D" parent="InnerDivider"]'), [BAY, NEG]),
    ("MV6 inner divider south disabled", SCENE, *disable('[node name="CollisionShape2D_2" type="CollisionShape2D" parent="InnerDivider"]'), [BAY, NEG]),
    ("MD1 west door trigger radius 6 -> 40", SCENE, '[sub_resource type="CircleShape2D" id="CircleShape2D_door_entry"]\nradius = 6.0',
     '[sub_resource type="CircleShape2D" id="CircleShape2D_door_entry"]\nradius = 40.0', [BAY]),
    ("MD2 arrival Spawn 1 tile from the door", SCENE, '[node name="Spawn" type="Marker2D" parent="door_entry_west"]\nposition = Vector2(32, 16)',
     '[node name="Spawn" type="Marker2D" parent="door_entry_west"]\nposition = Vector2(16, 8)', [BAY]),
    ("MC3 room bounds ignore metadata/room_shape", ROUTER, "\tif shape != null and shape.is_valid():\n\t\treturn shape.bounds_px()\n",
     "\tif false and shape != null and shape.is_valid():\n\t\treturn shape.bounds_px()\n", [BAY]),
    ("MK1 checkpoint spawn anchor removed", SCENE, 'spawn_anchor_id = &"ch01.maintenance_bay.safe_anchor_entry"\n', "", [BAY]),
    ("MR1 counterweight gear drifts +16 px (cluster)", SCENE,
     '[node name="counterweight_gear" type="Area2D" parent="."]\nposition = Vector2(512, 176)',
     '[node name="counterweight_gear" type="Area2D" parent="."]\nposition = Vector2(528, 176)', [BAY]),
]


def run(test):
    try:
        out = subprocess.run([GODOT, "--headless", "--path", str(COPY), "-s", "res://tests/run_tests.gd", "--", test],
                             capture_output=True, text=True, timeout=120, encoding="utf-8", errors="replace")
        text = out.stdout + out.stderr
    except subprocess.TimeoutExpired:
        return "TIMEOUT (120 s cap)", []
    m = re.search(r"TESTS: (\d+) passed, (\d+) failed", text)
    fails = []
    for l in text.splitlines():
        if l.strip().startswith("FAIL:"):
            fails.append(l.strip()[:220])
    return (m.group(0) if m else "no summary") + ", SCRIPT ERROR lines %d" % text.count("SCRIPT ERROR"), fails


only = sys.argv[2:] if len(sys.argv) > 2 else None
for name, rel, a, b, tests in MUTATIONS:
    if only and not any(name.startswith(o) for o in only):
        continue
    path = COPY / rel
    orig = path.read_bytes()
    text = orig.decode("utf-8")
    assert text.count(a) == 1, (name, a)
    path.write_bytes(text.replace(a, b).encode("utf-8"))
    try:
        results = [(t,) + run(t) for t in tests]
    finally:
        path.write_bytes(orig)
    print("MUTATION", name, flush=True)
    for t, summary, fails in results:
        print("  %s: %s" % (t.rsplit("/", 1)[1], summary), flush=True)
        for f in fails:
            print("    " + f, flush=True)
