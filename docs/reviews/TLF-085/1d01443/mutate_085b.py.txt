import re, subprocess, sys, pathlib

COPY = pathlib.Path(sys.argv[1])
GODOT = r"D:\Godot\Godot_v4.7.2-stable_win64_console.exe"
SCENE = "scenes/chapters/ch01/rooms/maintenance_bay.tscn"
ROUTER = "scripts/rooms/room_router.gd"
BAY = "res://tests/chapters/ch01/test_ch01_maintenance_bay_v2.gd"


def disable(node_header):
    return (node_header, node_header + "\ndisabled = true")


def block(name, pos):
    anchor = '[node name="TileMapLayer_Floor"'
    return (anchor, '[node name="%s" type="StaticBody2D" parent="."]\nposition = Vector2%s\n\n'
            '[node name="CollisionShape2D" type="CollisionShape2D" parent="%s"]\nshape = SubResource("CircleShape2D_door")\n\n%s'
            % (name, pos, name, anchor))


MUTATIONS = [
    ("MV1 bulkhead north segment disabled", SCENE, *disable('[node name="CollisionShape2D_1" type="CollisionShape2D" parent="Bulkhead"]'), [BAY]),

def run(test):
    try:
        out = subprocess.run([GODOT, "--headless", "--path", str(COPY), "-s", "res://tests/run_tests.gd", "--", test],
                             capture_output=True, text=True, timeout=120, encoding="utf-8", errors="replace")
        text = out.stdout + out.stderr
    except subprocess.TimeoutExpired:
        return "TIMEOUT (120 s cap)", []
    m = re.search(r"TESTS: (\d+) passed, (\d+) failed", text)
    fails = [l.strip()[:220] for l in text.splitlines() if l.strip().startswith("FAIL")]
    return (m.group(0) if m else "no summary") + ", SCRIPT ERROR lines %d" % text.count("SCRIPT ERROR"), fails


only = sys.argv[2:]
for name, rel, a, b, tests in MUTATIONS:
    if only and not any(name.startswith(o) for o in only):
        continue
    path = COPY / rel
    orig = path.read_bytes()
    text = orig.decode("utf-8")
    assert text.count(a) == 1, (name, text.count(a))
    path.write_bytes(text.replace(a, b, 1).encode("utf-8"))
    try:
        results = [(t,) + run(t) for t in tests]
    finally:
        path.write_bytes(orig)
    print("MUTATION", name, flush=True)
    for t, summary, fails in results:
        print("  %s: %s" % (t.rsplit("/", 1)[1], summary), flush=True)
        for f in fails[:8]:
            print("    " + f, flush=True)
