OUT = C.parent / "mutations_076b.log"
T = "\t"
N = "\n"

M = [
    ("L1 room swap leaks the old room (orphans)", "scripts/rooms/room_router.gd",

     + "func _reset_probe() -> void:" + N + T + "position = Vector2(600, 394)" + N + T + "_is_rewound = false" + N

def run():
    r = subprocess.run([GODOT, "--headless", "--path", str(C), "-s", "res://tests/run_some.gd"],
                       env={**os.environ, "TLF_ONLY": "test_ch01_mira_memory_side"},
                       capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=900)
    out = r.stdout + r.stderr
    tests = [l for l in out.splitlines() if l.startswith("TESTS:")]
    fails = [l.strip() for l in out.splitlines() if "FAIL:" in l]
    serr = sum(1 for l in out.splitlines() if "SCRIPT ERROR" in l)
    return tests[-1] if tests else "TESTS: (none) exit %d" % r.returncode, fails, serr


with OUT.open("w", encoding="utf-8") as log:
    t, f, s = run()
    log.write("BASELINE | %s | SCRIPT ERROR %d\n" % (t, s)); log.flush()
    for label, rel, a, b in M:
        path = C / rel
        orig = path.read_bytes()
        text = orig.decode("utf-8")
        if text.count(a) != 1:
            log.write("%s | PATCH NOT APPLIED (%d matches)\n" % (label, text.count(a)))
            continue
        path.write_bytes(text.replace(a, b).encode("utf-8"))
        try:
            t, f, s = run()
        finally:
            path.write_bytes(orig)
        verdict = "RED" if "0 failed" not in t else "GREEN (not caught)"
        log.write("%s | %s | %s | SCRIPT ERROR %d\n" % (label, verdict, t, s))
        for line in f[:4]:
            log.write("    %s\n" % line)
        log.flush()
    t, f, s = run()
    log.write("RESTORED | %s | SCRIPT ERROR %d\n" % (t, s))
print(OUT.read_text(encoding="utf-8"))
