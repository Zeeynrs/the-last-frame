def run():
    try:
        r = subprocess.run([GODOT, "--headless", "--path", str(C), "-s", "res://tests/run_tests.gd", "--", SUITE],
                           capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=120)
        out = r.stdout + r.stderr
    except subprocess.TimeoutExpired:
        out = "TIMEOUT 120 s"
    tests = [l for l in out.splitlines() if l.startswith("TESTS:")]
    fails = [l.strip() for l in out.splitlines() if "FAIL:" in l]
    serr = sum(1 for l in out.splitlines() if "SCRIPT ERROR" in l)
    return (tests[-1] if tests else "TESTS: (none) " + out[-60:].replace("\n", " ")), fails, serr


with OUT.open("w", encoding="utf-8") as log:
    t, f, s = run()
    log.write("BASELINE | %s | SCRIPT ERROR %d\n" % (t, s)); log.flush()
    for label, rel, a, b in M:
        path = C / rel
        orig = path.read_bytes()
        text = orig.decode("utf-8")
        if text.count(a) != 1:
            log.write("%s | PATCH NOT APPLIED (%d matches)\n" % (label, text.count(a))); log.flush()
            continue
        path.write_bytes(text.replace(a, b).encode("utf-8"))
        try:
            t, f, s = run()
        finally:
            path.write_bytes(orig)
        log.write("%s | %s | %s | SCRIPT ERROR %d\n" % (label, "RED" if "0 failed" not in t else "GREEN (not caught)", t, s))
        for line in f[:3]:
            log.write("    %s\n" % line)
        log.flush()
    t, f, s = run()
    log.write("RESTORED | %s | SCRIPT ERROR %d\n" % (t, s))
