"""TLF-080 cb1fbb1 re-check: 1 px module stubs (WallN_7..9, WallW_1), the conduit positions and
apply_state; Flash's concourse v2 suite must go RED. Renamed scratch copy, 120 s cap, files restored."""
import os, pathlib, subprocess

C = pathlib.Path(r"C:/Users/adity/AppData/Local/Temp/tlf080_c/copy")
GODOT = r"D:/Godot/Godot_v4.7.2-stable_win64_console.exe"
OUT = C.parent / "mutations_cb1fbb1.log"
SUITE = "res://tests/chapters/ch01/test_ch01_concourse_west_v2.gd"
SCENE = "scenes/chapters/ch01/rooms/concourse_west.tscn"
CP = "scripts/puzzle/objects/conduit_prop.gd"
T, N = "\t", "\n"


def wall(name, pos, new):
    head = '[node name="%s" type="Sprite2D" parent="YSort_Walls"]' % name
    return ("%s +1 px" % name, SCENE, head + N + "position = Vector2(%s)" % pos, head + N + "position = Vector2(%s)" % new)


M = [
    wall("WallN_7", "560, 184", "560, 185"),
    wall("WallN_8", "592, 200", "592, 201"),
    wall("WallN_9", "624, 216", "624, 217"),
    wall("WallW_1", "272, 88", "272, 89"),
    ("conduit fallen position +1 px", CP, "const POS_FALLEN := Vector2(328, 184)", "const POS_FALLEN := Vector2(328, 185)"),
    ("apply_state does not move the conduit", CP,
     T + 'conduit_state = StringName(state["conduit_state"])' + N + T + "_update_position()",
     T + 'conduit_state = StringName(state["conduit_state"])'),
]


def run():
    try:
        r = subprocess.run([GODOT, "--headless", "--path", str(C), "-s", "res://tests/run_tests.gd", "--", SUITE],
                           capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=120)
        out = r.stdout + r.stderr
    except subprocess.TimeoutExpired:
        out = "TIMEOUT 120 s"
    tests = [l for l in out.splitlines() if l.startswith("TESTS:")]
    fails = [l.strip() for l in out.splitlines() if "FAIL:" in l]
    serr = sum(1 for l in out.splitlines() if "SCRIPT ERROR" in l)
    return (tests[-1] if tests else "TESTS: (none) " + out[-60:].replace("\n", " ")), fails, serr


with OUT.open("w", encoding="utf-8") as log:
    t, f, s = run()
    log.write("BASELINE | %s | SCRIPT ERROR %d\n" % (t, s)); log.flush()
    for label, rel, a, b in M:
        path = C / rel
        orig = path.read_bytes()
        text = orig.decode("utf-8")
        if text.count(a) != 1:
            log.write("%s | PATCH NOT APPLIED (%d matches)\n" % (label, text.count(a))); log.flush()
            continue
        path.write_bytes(text.replace(a, b).encode("utf-8"))
        try:
            t, f, s = run()
        finally:
            path.write_bytes(orig)
        log.write("%s | %s | %s | SCRIPT ERROR %d\n" % (label, "RED" if "0 failed" not in t else "GREEN (not caught)", t, s))
        for line in f[:2]:
            log.write("    %s\n" % line)
        log.flush()
    t, f, s = run()
    log.write("RESTORED | %s | SCRIPT ERROR %d\n" % (t, s))
print(OUT.read_text(encoding="utf-8"))
