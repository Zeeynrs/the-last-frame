def run(test_file):
    try:
        r = subprocess.run([GODOT, "--headless", "--path", str(C), "-s", "res://tests/run_tests.gd", "--", test_file],
                           capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=120)
        out = r.stdout + r.stderr
    except subprocess.TimeoutExpired as e:
        out = (e.stdout or b"").decode("utf-8", "replace") if isinstance(e.stdout, bytes) else (e.stdout or "")
        out += "\nTIMEOUT 120 s"
    tests = [l for l in out.splitlines() if l.startswith("TESTS:")]
    fails = [l.strip() for l in out.splitlines() if "FAIL:" in l]
    serr = sum(1 for l in out.splitlines() if "SCRIPT ERROR" in l)
    return (tests[-1] if tests else "TESTS: (none)" + (" TIMEOUT" if "TIMEOUT" in out else "")), fails, serr

                for line in f[:4]:
                    log.write("    %s\n" % line)
                log.flush()
        finally:
            path.write_bytes(orig)
    t, f, s = run(SUITE)
    log.write("RESTORED suite | %s | SCRIPT ERROR %d\n" % (t, s))
