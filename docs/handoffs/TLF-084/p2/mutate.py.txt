def run(test: str) -> tuple:
    env = dict(os.environ, TLF_ONLY=test)
    try:
        out = subprocess.run([GODOT, "--headless", "--path", str(COPY), "-s", "res://tests/run_some.gd"],
                             capture_output=True, text=True, timeout=120, env=env, encoding="utf-8", errors="replace")
        text = out.stdout + out.stderr
    except subprocess.TimeoutExpired:
        return ("TIMEOUT", [])
    m = re.search(r"TESTS: (\d+) passed, (\d+) failed", text)
    fails = sorted({re.sub(r"^.*::", "", l.split("]")[0]) for l in text.splitlines() if l.strip().startswith("FAIL:")})
    errs = text.count("SCRIPT ERROR")
    return ((m.group(0) if m else "no summary") + f", SCRIPT ERROR lines {errs}", fails)


leak_decl = ("var _shade_tex: ImageTexture = null\n", "var _shade_tex: ImageTexture = null\nstatic var _leak: Array = []\n")
for name, rel, a, b in MUTATIONS:
    path = COPY / rel
    orig = path.read_text(encoding="utf-8")
    assert orig.count(a) == 1, (name, a)
    mutated = orig.replace(a, b)
    if name == "maps leak":
        mutated = mutated.replace(*leak_decl)
    path.write_text(mutated, encoding="utf-8", newline="\n")
    try:
        results = [(t,) + run(t) for t in ("test_shadows", "test_lighting_pass")]
    finally:
        path.write_text(orig, encoding="utf-8", newline="\n")
    print(f"MUTATION {name}")
    for t, summary, fails in results:
        print(f"  {t}: {summary}")
        for f in fails:
            print(f"    fails: {f}")
    sys.stdout.flush()
